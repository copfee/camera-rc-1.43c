cmake_minimum_required(VERSION 3.20)
option(RC_HOST_BUILD "Build portable core and tests without ESP-IDF" OFF)
if(RC_HOST_BUILD)
    project(camera_rc_tests LANGUAGES C CXX)
    enable_testing()
    add_subdirectory(tests)
else()
    set(PROJECT_VER "0.2.0")
    if(DEFINED IDF_TARGET AND NOT IDF_TARGET STREQUAL "esp32s3")
        message(FATAL_ERROR "This board requires IDF_TARGET=esp32s3")
    endif()
    set(IDF_TARGET "esp32s3" CACHE STRING "Fixed target for Waveshare 1.43C")
    include($ENV{IDF_PATH}/tools/cmake/project.cmake)
    project(camera_rc VERSION 0.2.0)
    # furble also patches NimBLE itself, not only its camera protocol code.
    idf_component_get_property(bt_lib bt COMPONENT_LIB)
    idf_component_get_property(bt_dir bt COMPONENT_DIR)
    set(RC_NIMBLE_SOURCE "${bt_dir}/host/nimble/nimble/nimble/host/src/ble_gap.c")
    set(RC_NIMBLE_OUTPUT "${CMAKE_BINARY_DIR}/furble_compat/ble_gap.c")
    include("${CMAKE_CURRENT_LIST_DIR}/tools/cmake/furble_nimble_patch.cmake")
    get_target_property(bt_sources ${bt_lib} SOURCES)
    list(FIND bt_sources "${RC_NIMBLE_SOURCE}" source_index)
    if(source_index EQUAL -1)
        message(FATAL_ERROR "NimBLE GAP source missing from bt component")
    endif()
    list(REMOVE_AT bt_sources ${source_index})
    list(APPEND bt_sources "${RC_NIMBLE_OUTPUT}")
    set_property(TARGET ${bt_lib} PROPERTY SOURCES "${bt_sources}")
    get_filename_component(nimble_src_dir "${RC_NIMBLE_SOURCE}" DIRECTORY)
    target_include_directories(${bt_lib} PRIVATE "${nimble_src_dir}")
    # Preserve the original encryption error instead of silently starting SMP
    # again after PINKEY_MISSING. Only an explicit pairing flow may create keys.
    target_compile_definitions(${bt_lib} PRIVATE MYNEWT_VAL_BLE_RESTART_PAIR=0)
endif()
